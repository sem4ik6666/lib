local FillCorner = Instance.new("UICorner"); FillCorner.CornerRadius = UDim.new(1, 0); FillCorner.Parent = Fill

local posX = input.Position.X - Track.AbsolutePosition.X; local ratio = math.clamp(posX / Track.AbsoluteSize.X, 0, 1); Fill.Size = UDim2.new(ratio, 0, 1, 0)
local rawVal = min + (max - min) * ratio; local val = math.floor(rawVal * 10) / 10; ValLabel.Text = tostring(val); configTable[configKey] = val; callback(val)
end
local function updateVisualDirect(val) local ratio = math.clamp((val - min) / (max - min), 0, 1); Fill.Size = UDim2.new(ratio, 0, 1, 0); ValLabel.Text = tostring(val) end
Track.InputBegan:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 then isDragging = true; update(input) end end)
UserInputService.InputEnded:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 then isDragging = false end end)
UserInputService.InputChanged:Connect(function(input) if isDragging and input.UserInputType == Enum.UserInputType.MouseMovement then update(input) end end)
table.insert(SlidersToUpdate, {Table = configTable, Key = configKey, UpdateFunc = updateVisualDirect, Callback = callback})
end

local function saveConfig(name)
local configData = {
ESP = { Enabled = ESPSettings.Enabled, CornerBox = ESPSettings.CornerBox, FullBox = ESPSettings.FullBox, HealthBar = ESPSettings.HealthBar, HealthText = ESPSettings.HealthText, NameESP = ESPSettings.NameESP, DistanceESP = ESPSettings.DistanceESP, MaxDistance = ESPSettings.MaxDistance },
Aim = { Enabled = AimSettings.Enabled, TeamCheck = AimSettings.TeamCheck, ShowFOV = AimSettings.ShowFOV, FOVSize = AimSettings.FOVSize, Smoothness = AimSettings.Smoothness }
}
if writefile then
writefile(ConfigFolderName .. "/" .. name .. ".json", HttpService:JSONEncode(configData))
Title.Text = "Custom Hub | Config [" .. name .. "] saved!"
task.delay(2, function() Title.Text = "Custom Hub | ESP & MOUSE CAMLOCK" end)
refreshConfigList()
end
end
local function loadConfig(name)
local filePath = ConfigFolderName .. "/" .. name .. ".json"
if readfile and isfile and isfile(filePath) then
local success, decoded = pcall(function() return HttpService:JSONDecode(readfile(filePath)) end)
if success and decoded then
if decoded.ESP then for k, v in pairs(decoded.ESP) do ESPSettings[k] = v end end
if decoded.Aim then for k, v in pairs(decoded.Aim) do AimSettings[k] = v end end
for _, t in ipairs(TogglesToUpdate) do t.UpdateFunc(t.Table[t.Key]); t.Callback(t.Table[t.Key]) end
for _, s in ipairs(SlidersToUpdate) do s.UpdateFunc(s.Table[s.Key]); s.Callback(s.Table[s.Key]) end
Title.Text = "Custom Hub | Config [" .. name .. "] loaded!"
task.delay(2, function() Title.Text = "Custom Hub | ESP & MOUSE CAMLOCK" end)
end
end
end

local texts = {obj.HealthText, obj.NameText, obj.DistanceText}

local function hideESP(obj)
obj.Box.Visible = false; obj.BoxOutline.Visible = false
for i = 1, 8 do obj.Corners[i].Visible = false end
obj.HealthBarBG.Visible = false; obj.HealthBar.Visible = false; obj.HealthText.Visible = false
obj.NameText.Visible = false; obj.DistanceText.Visible = false
end
-- Кэш игроков
local ActiveRenderCache = {}
local function setupCharacterEvents(plr, char)
if not char then return end
local hrp = char:WaitForChild("HumanoidRootPart", 5)
local head = char:WaitForChild("Head", 5)
local hum = char:WaitForChild("Humanoid", 5)
if hrp and head and hum then
ActiveRenderCache[plr] = { HRP = hrp, Head = head, Hum = hum, Dist = 0 }
end
end
local function onPlayerAdded(plr)
if plr == LocalPlayer then return end
plr.CharacterAdded:Connect(function(char) setupCharacterEvents(plr, char) end)
if plr.Character then setupCharacterEvents(plr, plr.Character) end
end
Players.PlayerAdded:Connect(onPlayerAdded)
for _, p in ipairs(Players:GetPlayers()) do onPlayerAdded(p) end
Players.PlayerRemoving:Connect(function(plr)
ActiveRenderCache[plr] = nil
if ESPCache[plr] then hideESP(ESPCache[plr]); ESPCache[plr] = nil end
end)
-- Поиск 2D цели для Аима
local function getClosest2DAimTarget()
local closestTargetPos = nil
local maxRadius = AimSettings.FOVSize
local mousePos = UserInputService:GetMouseLocation()
for plr, data in pairs(ActiveRenderCache) do
if data.Hum and data.Hum.Health > 0 and data.Head then
if AimSettings.TeamCheck and plr.Team == LocalPlayer.Team then continue end
local screenPos, onScreen = WorldToViewportPoint(Camera, data.Head.Position)
if onScreen then
local target2D = Vector2.new(screenPos.X, screenPos.Y)
local distanceToMouse = (target2D - mousePos).Magnitude
if distanceToMouse < maxRadius then
maxRadius = distanceToMouse
closestTargetPos = target2D
end
end
end
end
return closestTargetPos
end

local mousePos = UserInputService:GetMouseLocation()
if AimSettings.Enabled and AimSettings.ShowFOV then
FOVCircle.Position = mousePos
FOVCircle.Radius = AimSettings.FOVSize
FOVCircle.Visible = true
else
FOVCircle.Visible = false

if AimSettings.Enabled and UserInputService:IsMouseButtonPressed(AimSettings.Key) then
local target2D = getClosest2DAimTarget()
if target2D then
local mouseDeltaX = (target2D.X - mousePos.X) / AimSettings.Smoothness
local mouseDeltaY = (target2D.Y - mousePos.Y) / AimSettings.Smoothness
if mousemoverel then
mousemoverel(mouseDeltaX, mouseDeltaY)
end
end
end

for _, esp in pairs(ESPCache) do hideESP(esp) end
return
end

local sortableList = {}
for plr, data in pairs(ActiveRenderCache) do
if data.Hum and data.Hum.Health > 0 and data.HRP then
local distance = (camPos - data.HRP.Position).Magnitude
if distance <= ESPSettings.MaxDistance then
data.Dist = distance
table.insert(sortableList, {Plr = plr, Data = data})
else if ESPCache[plr] then hideESP(ESPCache[plr]) end end
else if ESPCache[plr] then hideESP(ESPCache[plr]) end end
end
table.sort(sortableList, function(a, b) return a.Data.Dist < b.Data.Dist end)
for i, item in ipairs(sortableList) do
local plr = item.Plr
local data = item.Data
local esp = ESPCache[plr]
if i > ESPSettings.MaxTargets then
if esp then hideESP(esp) end
continue
end
if not esp then esp = createESPObject(); ESPCache[plr] = esp end
local hrpPos, onScreen = WorldToViewportPoint(Camera, data.HRP.Position)
if not onScreen then hideESP(esp); continue end
local topPos = data.Head.Position + Vector3.new(0, 1.6, 0)
local bottomPos = data.HRP.Position - Vector3.new(0, 3.2, 0)
local top2D = WorldToViewportPoint(Camera, topPos)
local bottom2D = WorldToViewportPoint(Camera, bottomPos)
local height = math.abs(bottom2D.Y - top2D.Y)
local width = height * 0.65
local boxPos = Vector2.new(hrpPos.X - width / 2, top2D.Y)
local boxSize = Vector2.new(width, height)
if ESPSettings.FullBox then
esp.BoxOutline.Position = boxPos; esp.BoxOutline.Size = boxSize; esp.BoxOutline.Visible = true
esp.Box.Position = boxPos; esp.Box.Size = boxSize; esp.Box.Color = ESPSettings.BoxColor; esp.Box.Visible = true
else esp.Box.Visible = false; esp.BoxOutline.Visible = false end
if ESPSettings.CornerBox then
local lineLength = width * 0.25
local x, y = boxPos.X, boxPos.Y
local w, h = boxSize.X, boxSize.Y
local fromCoords = {
Vector2.new(x, y), Vector2.new(x, y), Vector2.new(x + w, y), Vector2.new(x + w, y),
Vector2.new(x, y + h), Vector2.new(x, y + h), Vector2.new(x + w, y + h), Vector2.new(x + w, y + h)
}
local toCoords = {
Vector2.new(x + lineLength, y), Vector2.new(x, y + lineLength), Vector2.new(x + w - lineLength, y), Vector2.new(x + w, y + lineLength),
Vector2.new(x + lineLength, y + h), Vector2.new(x, y + h - lineLength), Vector2.new(x + w - lineLength, y + h), Vector2.new(x + w, y + h - lineLength)
}
for j = 1, 8 do
local l = esp.Corners[j]; l.From = fromCoords[j]; l.To = toCoords[j]; l.Color = ESPSettings.BoxColor; l.Visible = true
end
else for j = 1, 8 do esp.Corners[j].Visible = false end end
if ESPSettings.HealthBar then
local barW = 3; local barX = boxPos.X - 6
local healthFactor = math.clamp(data.Hum.Health / data.Hum.MaxHealth, 0, 1)
esp.HealthBarBG.Position = Vector2.new(barX - 1, boxPos.Y - 1); esp.HealthBarBG.Size = Vector2.new(barW + 2, height + 2); esp.HealthBarBG.Visible = true
esp.HealthBar.Position = Vector2.new(barX, boxPos.Y + (height * (1 - healthFactor))); esp.HealthBar.Size = Vector2.new(barW, height * healthFactor); esp.HealthBar.Color = ESPSettings.HealthColor; esp.HealthBar.Visible = true
if ESPSettings.HealthText then
esp.HealthText.Text = tostring(math.floor(data.Hum.Health))
esp.HealthText.Position = Vector2.new(barX - 14, esp.HealthBar.Position.Y); esp.HealthText.Color = ESPSettings.HealthColor; esp.HealthText.Visible = true
else esp.HealthText.Visible = false end
else esp.HealthBarBG.Visible = false; esp.HealthBar.Visible = false; esp.HealthText.Visible = false end
if ESPSettings.NameESP then
esp.NameText.Text = plr.DisplayName or plr.Name
esp.NameText.Position = Vector2.new(boxPos.X + width / 2, boxPos.Y - 16); esp.NameText.Color = ESPSettings.TextColor; esp.NameText.Visible = true
else esp.NameText.Visible = false end
if ESPSettings.DistanceESP then
esp.DistanceText.Text = string.format("[%dm]", math.floor(data.Dist))
esp.DistanceText.Position = Vector2.new(boxPos.X + width / 2, boxPos.Y + height + 2); esp.DistanceText.Color = ESPSettings.TextColor; esp.DistanceText.Visible = true
else esp.DistanceText.Visible = false end
end
end)
